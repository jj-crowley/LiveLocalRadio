Route666.live nginx production config
=====================================

Copy the config below to your production server as something like:

  /etc/nginx/sites-available/route666.live

Then enable it with:

  sudo ln -s /etc/nginx/sites-available/route666.live /etc/nginx/sites-enabled/route666.live
  sudo nginx -t
  sudo systemctl reload nginx

This config assumes the app directory on the production server is:

  /var/www/route666

It also assumes the Next.js app is running locally on port 3000, usually through
the systemd service in:

  deploy/ubuntu-systemd.service

HTTPS note:

  If nginx is directly exposed to the internet, use the HTTPS config below.
  Port 80 only redirects visitors to HTTPS. You must install a TLS certificate
  first, for example with certbot.

  If you are using Cloudflare Tunnel and Cloudflare terminates HTTPS before
  traffic reaches this machine, your local nginx may intentionally listen on
  port 80 only. In that setup, Cloudflare handles the public 443 side.

nginx config
------------

server {
    listen 80;
    server_name route666.live www.route666.live;

    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    server_name route666.live www.route666.live;

    ssl_certificate /etc/letsencrypt/live/route666.live/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/route666.live/privkey.pem;

    # Artist submissions can include photos and multiple MP3 uploads.
    client_max_body_size 200m;

    # Serve uploaded photos and submitted files directly from disk in production.
    location /uploads/ {
        alias /var/www/route666/storage/uploads/;
        access_log off;
        expires 30d;
        add_header Cache-Control "public";
    }

    # Let Next.js resolve root-level artist block fallback paths.
    location /radio/_root/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }

    # Serve the Route666 streaming MP3 rotation directly from disk in production.
    location /radio/ {
        alias /var/www/route666/storage/radio/;
        access_log off;
        expires 1d;
        add_header Cache-Control "public";
        types {
            audio/mpeg mp3;
        }
    }

    # Let Next.js resolve masked video IDs before the static video alias.
    location /videos/id/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }

    # Serve local video files for embeddable players.
    location /videos/ {
        alias /var/www/route666/storage/videos/;
        access_log off;
        expires 1d;
        add_header Cache-Control "public";
        types {
            video/mp4 mp4;
            video/webm webm;
            video/ogg ogg ogv;
        }
    }

    location /_next/static/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
